<script setup>
import { isZh } from "../i18n";
import { ref } from "vue";
import {
  W95Button, W95Tooltip, W95Toast, W95Dropdown, W95MenuItem, W95ComboBox,
} from "vue-win95";

const ddOpen = ref(false);
const driver = ref(2);
const toasts = ref([{ id: 1, title: "Notepad", message: "Document saved.", duration: 0 }]);
const options = [
  { value: 1, label: "MySQL ODBC 3.51 Driver" },
  { value: 2, label: "SQL Server" },
];
let n = 1;
function pushToast() {
  toasts.value = [...toasts.value, { id: ++n, title: "System", message: "Hello from Windows 95." }];
}
</script>

<div v-if="isZh">

# Overlays 浮层

Tooltip · Dropdown · ComboBox · Toast

</div>
<div v-else>

# Overlays

Tooltip · Dropdown · ComboBox · Toast

</div>

<div class="demo">
  <div class="demo-label">TOOLTIP · DROPDOWN · COMBO · TOAST</div>
  <div style="display:flex;gap:10px;flex-wrap:wrap;align-items:center;padding:8px 0">
    <W95Tooltip content="Performs the default action">
      <W95Button default>Hover</W95Button>
    </W95Tooltip>
    <W95Dropdown v-model="ddOpen" :label="isZh ? '文件' : 'File'">
      <W95MenuItem>New</W95MenuItem>
      <W95MenuItem>Open…</W95MenuItem>
    </W95Dropdown>
    <W95ComboBox v-model="driver" :options="options" :label="isZh ? '驱动' : 'Driver'" />
    <W95Button @click="pushToast">Toast</W95Button>
  </div>
  <W95Toast v-model="toasts" />
</div>

<div v-if="isZh">
