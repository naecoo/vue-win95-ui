<script setup>
import { ref } from "vue";
import {
  W95Accordion,
  W95AccordionItem,
  W95Breadcrumb,
  W95Pagination,
  W95Switch,
  W95Avatar,
  W95GroupBox,
  W95Checkbox,
  W95Divider,
} from "vue-win95";

const open = ref("a");
const page = ref(1);
const sound = ref(true);

  { label: "Desktop" },

  { label: "C:" },
];

# Misc 其它组件

Accordion、Breadcrumb、Pagination、Switch、Avatar 与表单辅助件。

## Accordion

折叠面板，`aria-expanded` + region。

<div class="demo">
  <div class="demo-label">ACCORDION</div>
  <W95Accordion v-model="open">
    <W95AccordionItem value="a" title="General">System settings.</W95AccordionItem>
    <W95AccordionItem value="b" title="Appearance">Theme &amp; display.</W95AccordionItem>
    <W95AccordionItem value="c" title="Advanced">Power user options.</W95AccordionItem>
  </W95Accordion>
</div>

## Breadcrumb

`nav` + 列表，当前页 `aria-current="page"`。

<div class="demo">
  <div class="demo-label">BREADCRUMB</div>
  <W95Breadcrumb :items="crumbs" />
</div>
