## 全局安装

```ts
import { createApp } from "vue";
import Win95 from "vue-win95";
import "vue-win95/styles.css";

createApp(App).use(Win95).mount("#app");
```

## 按需引入（推荐）

```vue
<script setup lang="ts">
import { W95Button, W95Window } from "vue-win95";
import "vue-win95/styles.css";
</script>
```

样式只需引入一次 `vue-win95/styles.css`。

## 自动导入

```bash
pnpm add -D unplugin-vue-components
```

```ts
import Components from "unplugin-vue-components/vite";
import { VueWin95Resolver } from "@win95/vue-resolver";

export default {
  plugins: [Components({ resolvers: [VueWin95Resolver()] })],
};
```
