<script setup>
import { ref } from "vue";
import {
  W95Table,
  W95SpinButton,
  W95DatePicker,
  W95ProgressBar,
} from "vue-win95";

const selected = ref(null);
const copies = ref(2);
const date = ref("2024-06-15");
const progress = ref(45);

const columns = [
  { key: "name", label: "Name", width: 160 },
  { key: "version", label: "Version", width: 80 },
  { key: "company", label: "Company" },
];
const rows = [
  { id: 1, name: "MySQL ODBC 3.51", version: "3.51.11", company: "MySQL AB" },
  { id: 2, name: "SQL Server", version: "3.70.06", company: "Microsoft" },
  { id: 3, name: "Access Driver", version: "4.00", company: "Microsoft" },
];
</script>

# Data 数据组件

Table / SpinButton / DatePicker / ProgressBar。

## Table

可选中行的列表视图，贴合 Win95 Explorer / 控制面板风格。

<div class="demo">
  <div class="demo-label">TABLE</div>
  <W95Table
    v-model:selected-key="selected"
    :columns="columns"
    :rows="rows"
    :height="120"
  />
  <p style="color:#fff;font-size:13px;margin:8px 0 0">Selected: {{ selected }}</p>
</div>

## SpinButton

`role="spinbutton"`，支持 ↑↓ 与按钮步进，带 min/max 钳制。

<div class="demo">
  <div class="demo-label">SPIN BUTTON</div>
  <W95SpinButton v-model="copies" :min="0" :max="10" label="Copies" />
  <p style="color:#fff;font-size:13px;margin:8px 0 0">copies = {{ copies }}</p>
</div>

## DatePicker

月历对话框，输出 `YYYY-MM-DD`。

<div class="demo">
  <div class="demo-label">DATE PICKER</div>
  <W95DatePicker v-model="date" label="Date" />
  <p style="color:#fff;font-size:13px;margin:8px 0 0">date = {{ date }}</p>
</div>

## ProgressBar

`role="progressbar"` + `aria-valuenow`，支持分段样式。

<div class="demo">
  <div class="demo-label">PROGRESS</div>
  <W95ProgressBar :value="progress" label="Install" />
  <div style="margin-top:8px">
    <W95ProgressBar :value="progress" segmented label="Segmented" />
  </div>
</div>
