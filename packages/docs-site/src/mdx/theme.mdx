<script setup>
import { ref, watchEffect } from "vue";
import { W95Window, W95Button, W95Input, W95Slider } from "vue-win95";

const surface = ref("#c0c0c0");
const blue = ref("#000080");
const blueLight = ref("#1084d0");
const text = ref("#222222");
const desktop = ref("#008080");

watchEffect(() => {
  const r = document.documentElement.style;
  r.setProperty("--w95-surface", surface.value);
  r.setProperty("--w95-dialog-blue", blue.value);
  r.setProperty("--w95-dialog-blue-light", blueLight.value);
  r.setProperty("--w95-text", text.value);
  r.setProperty("--w95-desktop", desktop.value);
});

function reset() {
  surface.value = "#c0c0c0";
  blue.value = "#000080";
  blueLight.value = "#1084d0";
  text.value = "#222222";
  desktop.value = "#008080";
}
</script>
